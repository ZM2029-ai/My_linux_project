find_package(Threads REQUIRED)
find_path(MOSQUITTO_INCLUDE_DIR mosquitto.h)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp)
if(CMAKE_CROSSCOMPILING)
    # Debian multiarch：交叉构建不能误用 amd64 链接库。
    find_library(MOSQUITTO_LIBRARY mosquitto PATHS /usr/lib/arm-linux-gnueabihf NO_DEFAULT_PATH)
else()
    find_library(MOSQUITTO_LIBRARY mosquitto)
endif()
if(NOT MOSQUITTO_INCLUDE_DIR OR NOT MOSQUITTO_LIBRARY OR NOT JSON_INCLUDE_DIR)
    message(FATAL_ERROR "MQTT 需要 libmosquitto-dev（交叉构建为 :armhf）和 nlohmann-json3-dev")
endif()

add_library(edgelink_device_state STATIC device_message.cpp device_registry.cpp)
target_include_directories(edgelink_device_state PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_include_directories(edgelink_device_state SYSTEM PRIVATE ${JSON_INCLUDE_DIR})
target_compile_options(edgelink_device_state PRIVATE -Wall -Wextra)

add_library(edgelink_mqtt_core STATIC mqtt_client.cpp mqtt_gateway.cpp)
target_include_directories(edgelink_mqtt_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
target_include_directories(edgelink_mqtt_core SYSTEM PRIVATE ${MOSQUITTO_INCLUDE_DIR})
target_link_libraries(edgelink_mqtt_core PUBLIC edgelink_device_state ${MOSQUITTO_LIBRARY} Threads::Threads)
if(EDGELINK_ENABLE_SQLITE)
    target_link_libraries(edgelink_mqtt_core PRIVATE edgelink_device_store)
    target_compile_definitions(edgelink_mqtt_core PRIVATE EDGELINK_ENABLE_SQLITE)
endif()
target_compile_options(edgelink_mqtt_core PRIVATE -Wall -Wextra)

add_executable(edgelink_mqtt mqtt_main.cpp)
target_link_libraries(edgelink_mqtt PRIVATE edgelink_mqtt_core)
install(TARGETS edgelink_mqtt RUNTIME DESTINATION bin)

if(EDGELINK_BUILD_MQTT_CHECKS)
    add_executable(mqtt_state_check ../../tests/mqtt_state_check.cpp)
    target_link_libraries(mqtt_state_check PRIVATE edgelink_device_state)
    add_test(NAME mqtt_state_check COMMAND mqtt_state_check)
    add_executable(mqtt_client_check ../../tests/mqtt_client_check.cpp mqtt_client.cpp)
    target_include_directories(mqtt_client_check PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/.. ${MOSQUITTO_INCLUDE_DIR})
    target_compile_definitions(mqtt_client_check PRIVATE LIBMOSQUITTO_STATIC)
    target_link_libraries(mqtt_client_check PRIVATE Threads::Threads)
    add_test(NAME mqtt_client_check COMMAND mqtt_client_check)
    set_tests_properties(mqtt_client_check PROPERTIES TIMEOUT 30)
endif()
