find_package(Threads REQUIRED)
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp)
if(NOT JSON_INCLUDE_DIR)
    message(FATAL_ERROR "HTTP 服务需要 nlohmann-json3-dev")
endif()
add_executable(edgelink_http http_main.cpp)
target_include_directories(edgelink_http PRIVATE ${PROJECT_SOURCE_DIR}/vendor/cpp-httplib ${JSON_INCLUDE_DIR})
target_link_libraries(edgelink_http PRIVATE edgelink_device_store Threads::Threads)
if(EDGELINK_ENABLE_CONTROL)
    find_path(MOSQUITTO_INCLUDE_DIR mosquitto.h)
    if(CMAKE_CROSSCOMPILING)
        find_library(MOSQUITTO_LIBRARY mosquitto PATHS /usr/lib/arm-linux-gnueabihf NO_DEFAULT_PATH)
    else()
        find_library(MOSQUITTO_LIBRARY mosquitto)
    endif()
    if(NOT MOSQUITTO_INCLUDE_DIR OR NOT MOSQUITTO_LIBRARY)
        message(FATAL_ERROR "HTTP 远程控制需要 libmosquitto-dev（交叉构建为 :armhf）")
    endif()
    target_sources(edgelink_http PRIVATE control_service.cpp auth_config.cpp)
    target_include_directories(edgelink_http PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/.. ${MOSQUITTO_INCLUDE_DIR})
    target_link_libraries(edgelink_http PRIVATE ${MOSQUITTO_LIBRARY})
    target_compile_definitions(edgelink_http PRIVATE EDGELINK_ENABLE_CONTROL)
endif()
if(WIN32)
    target_link_libraries(edgelink_http PRIVATE ws2_32)
endif()
target_compile_options(edgelink_http PRIVATE -Wall -Wextra)
install(TARGETS edgelink_http RUNTIME DESTINATION bin)
